require("dotenv").config(); const express = require("express"); const path = require("path"); const OpenAI = require("openai"); const app = express(); const PORT = process.env.PORT || 3000; const client = new OpenAI({ apiKey: process.env.OPENAI_API_KEY }); app.use(express.json({ limit: "1mb" })); app.use(express.static(__dirname)); app.get("/api/health", (req, res) => { res.json({ ok: true, servidor: "funcionando", openai_configurado: !!process.env.OPENAI_API_KEY }); }); app.post("/api/chat", async (req, res) => { try { const { messages, // Recibimos el arreglo completo que manda tu index.html profile, mode } = req.body; if (!messages || !Array.isArray(messages) || messages.length === 0) { return res.status(400).json({ error: "No se recibió ningún mensaje." }); } const systemPrompt = { role: "system", content: ` Eres Adriana, una asistente virtual de apoyo psicológico y psicoeducación. Perfil: ${profile || "Adriana"} Modo: ${mode || "Conversar"} Habla en español, con un tono cálido, respetuoso y sencillo. Tu función es acompañar y ofrecer información psicológica general y ejercicios sencillos. No diagnostiques ni afirmes que sustituyes a un psicólogo, psiquiatra u otro profesional. No asegures que un síntoma físico es ansiedad o pánico. Si aparecen señales de una posible emergencia médica o riesgo inmediato, recomienda buscar ayuda profesional o servicios de emergencia locales. Responde de manera clara y breve, haciendo preguntas apropiadas cuando necesites comprender mejor la situación. ` }; // Preparamos los mensajes para OpenAI incluyendo el rol del sistema const formattedMessages = [ systemPrompt, ...messages.map(m => ({ role: m.role, content: m.content })) ]; const response = await client.chat.completions.create({ model: "gpt-4o-mini", messages: formattedMessages, temperature: 0.7 }); const reply = response.choices?.[0]?.message?.content || "No pude generar una respuesta en este momento."; res.json({ reply: reply }); } catch (error) { console.error("Error en /api/chat:", error); res.status(500).json({ error: "No se pudo comunicar con la IA." }); } }); app.listen(PORT, () => { console.log(""); console.log("================================="); console.log(" ADRIANA IA - SERVIDOR"); console.log("================================="); console.log(`Servidor: http://localhost:${PORT}`); console.log(`Salud: http://localhost:${PORT}/api/health`); console.log("================================="); });
